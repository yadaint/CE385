DATABASE

2. ฐานข้อมูล (prisma/schema.prisma)

มีทั้งหมด 8 models — ทุกความสัมพันธ์กำหนด onDelete: Cascade ลบพ่อแล้วลูกถูกลบตามอัตโนมัติ และฟิลด์ตัวตนสำคัญถูกบังคับ @unique ที่ระดับฐานข้อมูล

User — บัญชีผู้ใช้และแอดมิน (ตารางเดียวแยกด้วย role)

model User {
  id              String   @id @default(cuid())
  email           String   @unique            // ห้ามซ้ำ — ระดับฐานข้อมูล
  password        String                       // bcrypt hash 12 rounds
  role            Role     @default(USER)      // USER | ADMIN
  firstName       String?
  lastName        String?
  phone           String?  @unique            // เบอร์โทรใช้ซ้ำไม่ได้
  dateOfBirth     DateTime?
  nationalId      String?  @unique            // เลขหน้าบัตร 13 หลัก
  nationalIdBack  String?  @unique            // Laser code หลังบัตร
  employeeId      String?  @unique            // รหัสพนักงาน (เฉพาะแอดมิน) — ใช้สร้างอีเมล admin{รหัส}@domain
  bookings        Booking[]
  reviewedPayments Payment[] @relation("ReviewedPayments")
  staffDuties     StaffDuty[]
}
ทำไมบังคับ unique ที่ DB ไม่ใช่แค่โค้ด? ถ้าตรวจเฉพาะใน controller คนสองคนกดสมัครพร้อมกัน (มิลลิวินาทีเดียวกัน) จะผ่านการตรวจทั้งคู่แล้วเก็บซ้ำได้ — คอลัมน์ UNIQUE ของ PostgreSQL ปฏิเสธเองทันทีแม้พร้อมกัน (ระบบจะโยน Prisma error P2002 ซึ่ง error middleware แปลงเป็น 409 ให้)

Booking + BookingSeat — 1 ที่นั่ง = 1 รหัสตั๋ว

model Booking {
  id         String        @id @default(cuid())
  userId     String                        // ใครจอง (Cascade ลบตามเจ้าของ)
  concertId  String
  zoneId     String
  showDate   DateTime                      // วันแสดงที่เลือกจอง — ใช้ตัดสิน "บัตรหมดอายุ"
  ticketCode String @unique                  // รหัสประจำการจอง (= รหัสที่นั่งแรก)
  status     BookingStatus @default(PENDING)
  seats      BookingSeat[]
}
model BookingSeat {
  bookingId  String
  seatId     String
  ticketCode String? @unique               // ★ รหัสตั๋วประจำที่นั่ง — 1 ที่นั่ง = 1 รหัส
  @@id([bookingId, seatId])                 // PK คู่ — ที่นั่งเดียวอยู่ได้ 1 การจอง
}

สองระดับของรหัส: Booking.ticketCode ใช้อ้างอิงการจองทั้งรอบ ส่วน BookingSeat.ticketCode คือบัตรรายที่นั่งที่พิมพ์บนบัตรจริง (หน้าบัตรจะรวมทุกในมาแสดงบนบัตรเดียว)

StaffDuty — บันทึกเวรแอดมิน (ตารางใหม่)

model StaffDuty {
  id        String    @id @default(cuid())
  staffId   String                    // → User (Cascade)
  workDate  DateTime  @db.Date        // วันที่ทำงาน (ตัดเวลา)
  startedAt DateTime  @default(now())
  endedAt   DateTime?                 // null = เวรเปิดอยู่ (กำลังทำงานอยู่ตอนนี้)
}

หลักการ: แถวที่ endedAt = null คือเวรที่กำลังเปิดอยู่ — เมื่อแอดมินคนใหม่ล็อกอิน ระบบจะปิดเวรเก่า (กรอก endedAt) แล้วเปิดแถวใหม่ให้คนนั้นใน transaction เดียว จึงมีเวรเปิดได้ครั้งละ 1 เสมอ

4.1 register — สมัครสมาชิก (ได้ USER เสมอ)

ลำดับการทำงานสำคัญมาก — ตรวจข้อมูลซ้ำก่อนเข้ารหัสรหัสผ่าน เพื่อไม่เสียเวลาคำนวณ bcrypt ฟรี:

export const register = asyncHandler(async (req, res) => {
  const { email, password, ..., nationalId, nationalIdBack } = req.body;

  // 1) เช็คอีเมลซ้ำ → 409
  const existingUser = await prisma.user.findUnique({ where: { email } });
  if (existingUser) throw new AppError("อีเมลนี้ถูกใช้งานแล้ว", 409);

  // 2) จัดรูปแบบบัตร — ตัดตัวคั่น/พิมพ์ใหญ่ เพื่อให้ unique เทียบกันได้แม่นยำ
  const normalizedNationalId    = nationalId.replace(/\D/g, "");
  const normalizedNationalIdBack = nationalIdBack.replace(/[-\s]/g, "").toUpperCase();

  // 3) เช็คเบอร์/บัตรหน้า/บัตรหลัง พร้อมกัน (Promise.all — เร็วขึ้น 3 เท่า)
  const [existingPhone, existingNationalId, existingNationalIdBack] =
    await Promise.all([
      prisma.user.findFirst({ where: { phone } }),
      prisma.user.findFirst({ where: { nationalId: normalizedNationalId } }),
      prisma.user.findFirst({ where: { nationalIdBack: normalizedNationalIdBack } }),
    ]);
  // → ข้อความ error บอกด้วยว่า "ต้องลบบัญชีเดิมออกก่อน"

  // 4) ผ่านหมด → hash รหัสผ่าน + สร้างบัญชี role USER
  const user = await prisma.user.create({
    data: { email, password: await bcrypt.hash(password, 12), ..., role: "USER" },
  });
  res.status(201).json({ success: true, data: { user } });
});
จุดออกแบบ: role ไม่รับจาก request body — บังคับ USER เสมอ ทำให้ไม่มีใครสมัครเป็นแอดมินเองได้ และการเช็คซ้ำใช้ค่า normalize แล้ว (บัตร `1-2345-67890-12-3` กับ `1234567890123` คือเลขเดียวกัน)

4.3 adminLogin — อีเมลกลาง + รหัสผ่านส่วนตัว (หัวใจของหลังบ้าน)

ทุกแอดมินใช้อีเมลเดียวกัน (ADMIN_LOGIN_EMAIL) ตัวตนจริงระบุด้วยรหัสผ่าน — ระบบจึงต้องเทียบรหัสผ่านกับแอดมินทุกคน:

export const adminLogin = asyncHandler(async (req, res) => {
  const { email, password } = req.body;

  // 1) ต้องใช้อีเมลกลางเท่านั้น — อีเมลส่วนตัวใช้ไม่ได้
  if (email.toLowerCase() !== env.ADMIN_LOGIN_EMAIL.toLowerCase())
    throw new AppError("แอดมินต้องเข้าสู่ระบบด้วยอีเมลกลาง: ...", 401);

  // 2) เทียบรหัสผ่านกับแอดมินทุกคน — ตรงคนไหน = ล็อกอินเป็นคนนั้น
  const admins = await prisma.user.findMany({ where: { role: "ADMIN" } });
  let matched = null, duplicateCount = 0;
  for (const admin of admins) {
    if (await bcrypt.compare(password, admin.password)) {
      if (matched) duplicateCount += 1;   // รหัสตรงมากกว่า 1 คน → ระบุตัวตนไม่ได้
      else matched = admin;
    }
  }
  if (!matched)         throw new AppError("รหัสผ่านไม่ถูกต้อง...", 401);
  if (duplicateCount)   throw new AppError("รหัสผ่านนี้ซ้ำกับแอดมินมากกว่า 1 คน...", 409);

  // 3) ล็อกอิน = เริ่มเวรทันที (ปิดเวรค้าง + เปิดเวรใหม่ ใน transaction เดียว)
  const dutyNow = new Date();
  const workDate = new Date(Date.UTC(dutyNow.getFullYear(), dutyNow.getMonth(), dutyNow.getDate(), 12));
  await prisma.$transaction([
    prisma.staffDuty.updateMany({ where: { endedAt: null }, data: { endedAt: dutyNow } }),
    prisma.staffDuty.create({ data: { staffId: user.id, workDate, startedAt: dutyNow } }),
  ]);

  const token = generateToken({ sub: user.id, email: user.email, role: user.role });
  // → token ฝัง role ของ "คนที่รหัสตรง" ไม่ใช่เมลกลาง
});
เกร็ด 3 ข้อ: ① ตรวจรหัสผ่านก่อนเช็ค role เสมอ — ไม่รั่วว่าอีเมลไหนเป็นแอดมิน ② เวลา `workDate` ใช้เที่ยงวัน UTC (ชั่วโมง 12) เพื่อให้ตัดเป็น DATE ถูกวันแม้ timezone เพี้ยน ③ รหัสผ่านซ้ำถูกกันตั้งแต่ตอนสร้างบัญชี อีกชั้นตอนล็อกอิน (409)

BACKEND · BOOKING

5. Flow การจอง — Transaction ระดับ SERIALIZABLE

backend/src/controllers/booking.controller.tsbackend/src/utils/seats.ts

ปัญหาคลาสสิกของระบบจอง: สองคนกดจองที่นั่งเดียวกันพร้อมกัน ถ้าอ่าน-เขียนแบบธรรมดาจะได้ที่นั่งเดียวกันทั้งคู่ ระบบนี้จึงรันทุกขั้นตอนการจองใน $transaction ระดับ isolation SERIALIZABLE (PostgreSQL จะเลิกทำรายการที่ชนกันเอง) พร้อม retry อัตโนมัติเมื่อโดน error P2034:

const performBookingTransaction = (userId, seatIds, showDate) =>
  prisma.$transaction(async (tx) => {
    // 1) โหลดที่นั่งที่ขอ — ขาดตัวใด = 404
    const seats = await tx.seat.findMany({ where: { id: { in: seatIds } }, include: { zone: true } });
    // 2) ทุกที่นั่งต้องโซนเดียวกัน + ว่างทั้งหมด → 409 ถ้าขัดแย้ง
    // 3) ถือที่นั่ง:  isAvailable = false
    // 4) หัก availableSeats ของโซน และแจ้งเตือน "โซนหมด/คอนหมด" อัตโนมัติ
    // 5) สร้าง Booking + BookingSeat พร้อมรหัสตั๋วรายที่นั่ง
  }, { isolationLevel: Prisma.TransactionIsolationLevel.Serializable, timeout: 15_000 });

การสร้างรหัสตั๋วรายที่นั่ง + กันซ้ำ 5 รอบ

// รูปแบบ: {THA|NAT} + รหัสโซน + เลขที่นั่ง 2 หลัก + สุ่ม 3 ตัว = 10 ตัวอักษร
const bookedSeats = [...seats].sort((a, b) => a.seatNumber - b.seatNumber);
let seatTicketCodes = bookedSeats.map((seat) =>
  generateTicketCode(concert.origin, zone.code, seat.seatNumber));

for (let round = 0; round < 5; round++) {
  const seatClash   = await tx.bookingSeat.findMany({ where: { ticketCode: { in: seatTicketCodes } } });
  const bookingClash = await tx.booking.findMany({ where: { ticketCode: { in: seatTicketCodes } } });
  const used = new Set([...seatClash, ...bookingClash].map((row) => row.ticketCode));
  if (used.size === 0) break;
  seatTicketCodes = seatTicketCodes.map((code) =>
    used.has(code) ? generateTicketCode(concert.origin, zone.code, Number(code.slice(5, 7))) : code);
}

จากนั้นสร้าง Booking โดย ticketCode: seatTicketCodes[0] (รหัสอ้างอิงการจอง) และแต่ละ BookingSeat ได้รหัสของตัวเอง — หน้าบัตรจึงแสดงได้ครบทุกใน

11.2 VAT 7% — ราคารวมภาษี แยกย่อยออกมาแสดง

frontend/src/utils/format.ts
export const VAT_RATE = 0.07;
export const calculateVat = (totalWithVat: number): VatBreakdown => {
  const total     = Math.round(totalWithVat * 100) / 100;
  const beforeVat = Math.round((total / (1 + VAT_RATE)) * 100) / 100;   // ย้อนหายอดก่อน VAT
  return { beforeVat, vat: Math.round((total - beforeVat) * 100) / 100, total };
};

บัตรคอนเสิร์ตและหน้าชำระเงินจะแสดง 3 บรรทัด: ยอดก่อน VAT (7%) → ภาษีมูลค่าเพิ่ม (VAT 7%) → ยอดรวม พร้อมหมายเหตุ "ราคารวม VAT 7% แล้ว" — คำนวณจากฟังก์ชันกลางเดียว ตัวเลขตรงกันทุกจุด

frontend/src/pages/MyBookingsPage.tsx

โซนอันตรายท้ายหน้า "ตั๋วของฉัน" — กดปุ่มลบบัญชีไม่ลบทันที แต่เปิด ConfirmDialog ที่มีหมายเหตุว่า "ข้อมูลจะถูกลบถาวร + หากยังมีบัตรไม่หมดอายุ ระบบจะไม่อนุญาตให้ลบ" — กดยืนยันจึงเรียก DELETE /auth/me:

const handleDeleteAccount = async (): Promise<void> => {
  setIsDeletingAccount(true);
  try {
    await deleteAccount();                       // ถ้ามีบัตรค้าง → backend 409 พร้อมข้อความบอกจำนวน
    navigate("/", { replace: true });
  } catch (err) {
    setError(getApiErrorMessage(err));           // โชว์เหตุผลที่ลบไม่ได้บนหน้าเว็บ
    setIsDeleteDialogOpen(false);
  } finally { setIsDeletingAccount(false); }
};

เงื่อนไข "บัตรหมดอายุ" ตัดสินที่ backend (สถานะ ≠ CANCELLED และ showDate ≥ วันนี้) — ผู้ใช้เลี่ยงด้วยเครื่องมือใดไม่ได้ เพราะเป็นการนับจากฐานข้อมูลจริงทุกครั้ง

15. Seed ข้อมูลเดโม่ (prisma/seed.ts)

รันด้วย npm run prisma:seed — ข้ามส่วนที่มีข้อมูลอยู่แล้วอัตโนมัติ ถ้าอยากรีเซ็ตเดโม่ใหม่ทั้งหมด: รัน node reset-demo.cjs (ล้างคอน/การจอง) แล้ว seed ซ้ำ

  • แอดมิน 4 คน — อภิชาติ(0000)/Admin1234 · สมชาย(0001)/Admin0001 · สมหญิง(0002)/Staff1234 · มาลี(0003)/Admin0003 — รหัสผ่านทุกคนไม่ซ้ำกัน (เงื่อนไขอีเมลกลาง)
  • ลูกค้า 5 คน — user@concerthub.dev · somchai · malee · prakan · nok (รหัสผ่าน User1234 ทั้งหมด, โปรไฟล์ครบ)
  • คอนเสิร์ต 7 งาน — ครบ THAI/INTL × SOLO/GROUP + สถานะ OPEN 5 · DRAFT 1 · SOLD_OUT 1 (ซิมโฟนีออฟไฟร์เวิร์คส์ — ที่นั่งทุกโซนถือหมด)
  • การจอง 6 กรณีศึกษา — COMPLETED+สลิปอนุมัติ(ผู้ตรวจ admin0001) · PENDING_VERIFICATION · REJECTED(พร้อมเหตุผล) · PENDING · CANCELLED · จองบนคอนบัตรหมด — แต่ละรายการมี BookingSeat + รหัสตั๋วรายที่นั่งจริงตามรูปแบบ THAVP/A08...
รูปแบบรหัสตั๋วใน seed: {THA|NAT} + รหัสโซน + seat.code (แถว+เลข เช่น A08) + เลข 2 ตัว — ใช้ seat.code แทน seatNumber เพราะ seatNumber เริ่มนับ 1 ใหม่ทุกแถว จะซ้ำข้ามแถว (เช่น THAVPA0852, THAVPB0852)
